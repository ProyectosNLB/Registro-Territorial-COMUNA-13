function doPost(e) { var headers = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Access-Control-Max-Age": "86400" }; try { var data = JSON.parse(e.postData.contents); var spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); // NOTA: Reemplaza "Respuestas del formulario 1" por el nombre exacto de tu pestaña si es diferente var sheet = spreadsheet.getSheetByName("Respuestas del formulario 1") || spreadsheet.getActiveSheet(); // Procesar imágenes y guardarlas en Drive var urlsFotos = []; if (data.files && data.files.length > 0) { var carpeta = DriveApp.getRootFolder(); for (var i = 0; i < data.files.length; i++) { var archivoData = data.files[i]; var bytes = Utilities.base64Decode(archivoData.base64); var blob = Utilities.newBlob(bytes, "image/jpeg", "Foto_C13_" + archivoData.name); var archivoCreado = carpeta.createFile(blob); archivoCreado.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); urlsFotos.push(archivoCreado.getUrl()); } } // Forzamos la unión en formato de texto "latitud,longitud" var coordenadasConsolidadas = ""; if (data.lat && data.lng) { coordenadasConsolidadas = data.lat + "," + data.lng; } // Calcular la fila donde se va a escribir de forma absoluta var ultimaFila = sheet.getLastRow(); var filaDestino = ultimaFila + 1; // Insertamos una fila limpia al final sheet.insertRowsAfter(ultimaFila, 1); // ASIGNACIÓN ABSOLUTA COLUMNA POR COLUMNA sheet.getRange(filaDestino, 1).setValue(new Date()); // A: Fecha sheet.getRange(filaDestino, 2).setValue(data.nombre || ""); // B: Nombre sheet.getRange(filaDestino, 3).setValue(data.email || ""); // C: email sheet.getRange(filaDestino, 4).setValue(data.address || ""); // D: Dirección sheet.getRange(filaDestino, 5).setValue(data.barrio || ""); // E: Barrio sheet.getRange(filaDestino, 6).setValue("Comuna 13"); // F: Comuna sheet.getRange(filaDestino, 7).setValue(data.categoriaVecino || ""); // G: Categoría Vecino sheet.getRange(filaDestino, 8).setValue(data.descripcion || ""); // H: Descripción // Forzamos vacío en las columnas de la IA para que no se desplacen sheet.getRange(filaDestino, 9).setValue(""); // I: Categoría IA (Vacío) sheet.getRange(filaDestino, 10).setValue(""); // J: Prioridad IA (Vacío) sheet.getRange(filaDestino, 11).setValue(""); // K: Resumen IA (Vacío) // El carácter "'" fuerza a Google Sheets a interpretarlo como texto y no romperlo por la configuración regional sheet.getRange(filaDestino, 12).setValue("'" + coordenadasConsolidadas); // L: Coordenadas Juntas sheet.getRange(filaDestino, 13).setValue(urlsFotos.join("\n")); // M: Links Fotos return ContentService.createTextOutput(JSON.stringify({ "result": "success" })) .setMimeType(ContentService.MimeType.JSON) .setHeaders(headers); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })) .setMimeType(ContentService.MimeType.JSON) .setHeaders(headers); } } function doOptions(e) { return ContentService.createTextOutput("") .setMimeType(ContentService.MimeType.TEXT) .setHeaders({ "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Access-Control-Max-Age": "86400" }); }
